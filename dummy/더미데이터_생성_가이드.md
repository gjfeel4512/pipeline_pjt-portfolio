## 📝 더미 데이터 생성 스크립트

### 스크립트 1: `generate_dummy_bronze_data.py`

```python
"""
더미 YouTube 데이터 생성 스크립트
생성 위치: C:\Pipeline_pjt\outputs\bronze_merged\youtube_dummy_data.jsonl
"""

import json
import random
from datetime import datetime, timedelta
from pathlib import Path
import uuid

# 생성 설정
OUTPUT_DIR = Path("outputs/bronze_merged")
OUTPUT_FILE = OUTPUT_DIR / "youtube_dummy_data.jsonl"
NUM_RECORDS = 1000  # 생성할 레코드 수

# 더미 데이터용 샘플
CHANNELS = ["Tech Daily", "Coding Tutorials", "Music Channel", "Food Review", "Gaming"]
COUNTRIES = ["US", "KR", "JP", "CN", "GB", "DE", "FR", "CA", "AU", "BR"]
DEVICES = ["mobile", "desktop", "tablet"]

def generate_dummy_record(index):
    """단일 더미 레코드 생성"""
    
    # 랜덤 타임스탬프 (지난 30일)
    days_ago = random.randint(0, 30)
    occurred_at = datetime.utcnow() - timedelta(days=days_ago, hours=random.randint(0, 23))
    
    # 조회수 기반 통계 (로그 정규분포)
    view_count = int(10 ** random.uniform(1, 5))  # 10 ~ 100,000
    like_count = int(view_count * random.uniform(0.01, 0.1))
    comment_count = int(view_count * random.uniform(0.001, 0.05))
    
    # 동영상 길이 (초)
    duration_seconds = random.randint(60, 7200)  # 1분 ~ 2시간
    
    record = {
        "_id": str(uuid.uuid4()),
        "schema_version": "1.0",
        "record_type": "application_log",
        "event_id": str(uuid.uuid4()),
        "trace_id": str(uuid.uuid4()),
        "run_id": f"run_202609_{index:06d}",
        
        # 타임스탬프
        "occurred_at": occurred_at.isoformat() + "+09:00",
        "generated_at_utc": datetime.utcnow().isoformat() + "+00:00",
        
        # 도메인
        "domain": "youtube",
        "event_type": random.choice([
            "video_view",
            "video_search",
            "playlist_created",
            "video_uploaded",
            "comment_added",
            "like_clicked"
        ]),
        
        # 서비스 정보
        "service": {
            "name": "YouTube API",
            "version": "3.0",
            "region": "us-west-2"
        },
        
        # 클라이언트 정보
        "client": {
            "country": random.choice(COUNTRIES),
            "platform": random.choice(["iOS", "Android", "Web"]),
            "device_type": random.choice(DEVICES),
            "app_version": f"v{random.randint(1, 5)}.{random.randint(0, 9)}.{random.randint(0, 9)}"
        },
        
        # 요청 정보
        "request": {
            "method": "GET",
            "path": "/api/youtube/videos",
            "request_bytes": random.randint(100, 5000)
        },
        
        # 응답 정보
        "response": {
            "status_code": random.choice([200, 200, 200, 200, 201, 400, 429]),  # 대부분 성공
            "latency_ms": random.randint(50, 5000),
            "response_bytes": random.randint(1000, 50000)
        },
        
        # 비디오 데이터
        "data": {
            "video_id": f"vid_{index:08d}",
            "title": f"Sample Video {index}: {random.choice(['Tutorial', 'Review', 'Gaming', 'Music', 'Vlog'])}",
            "channel_name": random.choice(CHANNELS),
            "channel_id": f"ch_{random.randint(1000, 9999)}",
            
            # 메타데이터
            "duration": f"PT{duration_seconds // 60}M{duration_seconds % 60}S",
            "published_at": (occurred_at - timedelta(days=random.randint(1, 365))).isoformat(),
            
            # 통계
            "view_count": view_count,
            "like_count": like_count,
            "comment_count": comment_count,
            "share_count": int(view_count * random.uniform(0.001, 0.01)),
            "favorite_count": int(view_count * random.uniform(0.0001, 0.005)),
            
            # 추가 정보
            "is_monetized": random.choice([True, False, False]),
            "language": random.choice(["en", "ko", "ja", "zh", "es", "fr"]),
            "tags": [f"tag_{i}" for i in range(random.randint(3, 10))],
            
            # 품질 지표
            "engagement_rate": (like_count + comment_count) / view_count if view_count > 0 else 0
        }
    }
    
    return record

def generate_dummy_data(num_records=NUM_RECORDS):
    """더미 데이터 파일 생성"""
    
    print(f"📝 더미 데이터 생성 중... ({num_records}개 레코드)")
    print(f"💾 출력 위치: {OUTPUT_FILE}")
    
    # 출력 폴더 생성
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    
    # 데이터 생성 및 저장
    with open(OUTPUT_FILE, 'w', encoding='utf-8') as f:
        for i in range(num_records):
            record = generate_dummy_record(i)
            f.write(json.dumps(record, ensure_ascii=False) + '\n')
            
            # 진행 상황 표시
            if (i + 1) % 100 == 0:
                print(f"  ✓ {i + 1}/{num_records} 생성 완료")
    
    # 파일 크기 확인
    file_size = OUTPUT_FILE.stat().st_size / (1024 * 1024)  # MB
    
    print(f"\n✅ 데이터 생성 완료!")
    print(f"  - 파일: {OUTPUT_FILE}")
    print(f"  - 레코드: {num_records:,}개")
    print(f"  - 크기: {file_size:.2f} MB")
    print(f"\n다음 단계: 배포단계별_체크리스트.md의 11단계 진행")

if __name__ == "__main__":
    generate_dummy_data()
```
